cmake_minimum_required(VERSION 3.25)
project(HyperDRPatentProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(HYPERDR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(HYPERDR_CORE_BUILD "${HYPERDR_ROOT}/build-core" CACHE PATH "Built HyperDR core libraries")
add_executable(patent_probe probe.cpp)
target_compile_definitions(patent_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN HYPERDR_WITH_CODECS=0 HYPERDR_WITH_NCNN=0)
foreach(module gainmap look image container foundation)
  target_include_directories(patent_probe PRIVATE "${HYPERDR_ROOT}/modules/${module}/include")
  target_link_libraries(patent_probe PRIVATE "${HYPERDR_CORE_BUILD}/modules/${module}/$<CONFIG>/hyperdr_${module}.lib")
endforeach()
