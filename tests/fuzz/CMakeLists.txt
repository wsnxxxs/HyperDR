# Explicit opt-in; neither these executables nor libFuzzer enter a normal build.
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Fuzz targets require LLVM Clang and its libFuzzer runtime")
endif()
set(parser_targets heif_reader heif_inspect tmap exif hvcc cube)
set(parser_index 0)
foreach(parser IN LISTS parser_targets)
  math(EXPR parser_index "${parser_index} + 1")
  add_executable(fuzz_${parser} EXCLUDE_FROM_ALL parser_fuzz.cpp)
  target_compile_definitions(fuzz_${parser} PRIVATE HYPERDR_FUZZ_TARGET=${parser_index})
  target_compile_options(fuzz_${parser} PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz_${parser} PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_libraries(fuzz_${parser} PRIVATE HyperDR::container HyperDR::look)
endforeach()
